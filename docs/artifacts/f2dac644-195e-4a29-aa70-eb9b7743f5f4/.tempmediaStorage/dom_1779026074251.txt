import React, { useEffect, useState } from 'react'
import Sidebar from './components/Sidebar'
import DashboardView from './components/DashboardView'
import IngestView from './components/IngestView'
import TransactionsView from './components/TransactionsView'
import AnalysisView from './components/AnalysisView'
import ProjectionView from './components/ProjectionView'
import ReportsView from './components/ReportsView'
import PersonasView from './components/PersonasView'
import SettingsView from './components/SettingsView'
import AiChatPanel from './components/AiChatPanel'
import { ViewId, AppConfig } from './types'

function App(): JSX.Element {
  const [currentView, setCurrentView] = useState<ViewId>('dashboard')
  const [activePersonaId, setActivePersonaId] = useState<string | null>(null)
  const [config, setConfig] = useState<AppConfig | null>(null)

  useEffect(() => {
    loadAppConfig()
  }, [])

  const loadAppConfig = async () => {
    const cfg = await window.api.config.get() as AppConfig
    setConfig(cfg)
    setActivePersonaId(cfg.activePersonaId)
  }

  const handlePersonaChanged = () => {
    loadAppConfig()
  }

  const handleDataIngested = () => {
    // Jump directly to transactions review queue after ingestion so the user can verify
    setCurrentView('transactions')
  }

  return (
    <div className="app-container">
      {/* Sidebar Navigation */}
      <Sidebar 
        currentView={currentView}
        onViewChange={setCurrentView}
        activePersonaId={activePersonaId}
        onPersonaChanged={handlePersonaChanged}
      />

      {/* Main Workspace Area */}
      <main className="main-workspace">
        {currentView === 'dashboard' && (
          <DashboardView 
            activePersonaId={activePersonaId} 
            onViewChange={setCurrentView} 
          />
        )}

        {currentView === 'ingest' && (
          <IngestView 
            activePersonaId={activePersonaId} 
            onDataIngested={handleDataIngested} 
          />
        )}

        {currentView === 'transactions' && (
          <TransactionsView 
            activePersonaId={activePersonaId} 
          />
        )}

        {currentView === 'analysis' && (
          <AnalysisView 
            activePersonaId={activePersonaId} 
          />
        )}

        {currentView === 'projections' && (
          <ProjectionView 
            activePersonaId={activePersonaId} 
          />
        )}

        {currentView === 'reports' && (
          <ReportsView 
            activePersonaId={activePersonaId} 
          />
        )}

        {currentView === 'personas' && (
          <PersonasView 
            activePersonaId={activePersonaId}
            onPersonaChanged={handlePersonaChanged}
          />
        )}

        {currentView === 'settings' && (
          <SettingsView />
        )}

        <Versions />
      </main>

      {/* Floating pull-out AI Accountant Chat Panel */}
      <AiChatPanel activePersonaId={activePersonaId} />
    </div>
  )
}

function Versions(): JSX.Element {
  const [versions] = useState(window.electron.process.versions)

  return (
    <div className="versions-tag">
      Gemini Finance Edition â€¢ Electron v{versions.electron} â€¢ Node v{versions.node}
    </div>
  )
}

export default App
